def load_model(model_id: str, adapter_id: str = None, load_4bit: bool = True):
    """
    Load model từ HuggingFace. Ưu tiên unsloth, fallback transformers.
    Hỗ trợ LoRA adapter (safetensors).

    Returns: (model, tokenizer, served_name)
    """
    try:
        from unsloth import FastLanguageModel
        return _load_with_unsloth(model_id, adapter_id, load_4bit)
    except ImportError:
        print("⚠️  unsloth không có sẵn, dùng transformers thay thế.")
        return _load_with_transformers(model_id, adapter_id, load_4bit)


def _load_with_unsloth(model_id, adapter_id, load_4bit):
    from unsloth import FastLanguageModel

    print(f"📦 Loading model (unsloth): {model_id} (4-bit={load_4bit})")
    model, tokenizer = FastLanguageModel.from_pretrained(
        model_name=model_id,
        max_seq_length=16384,
        load_in_4bit=load_4bit,
    )

    served_name = model_id
    if adapter_id:
        from peft import PeftModel
        print(f"📦 Loading LoRA adapter (safetensors): {adapter_id}")
        model = PeftModel.from_pretrained(model, adapter_id)
        served_name = adapter_id
        print(f"  ✅ Adapter loaded: {adapter_id}")

    FastLanguageModel.for_inference(model)
    print(f"  ✅ Model ready! Serving as '{served_name}'")
    return model, tokenizer, served_name



def _load_with_transformers(model_id, adapter_id, load_4bit):
    import torch
    from transformers import AutoModelForCausalLM, AutoTokenizer, BitsAndBytesConfig

    print(f"📦 Loading tokenizer: {model_id}")
    tokenizer = AutoTokenizer.from_pretrained(model_id, trust_remote_code=True)
    if tokenizer.pad_token is None:
        tokenizer.pad_token = tokenizer.eos_token

    load_kwargs = {
        "trust_remote_code": True,
        "device_map": "auto",
        "torch_dtype": torch.float16,
    }
    if load_4bit:
        load_kwargs["quantization_config"] = BitsAndBytesConfig(
            load_in_4bit=True,
            bnb_4bit_compute_dtype=torch.float16,
            bnb_4bit_quant_type="nf4",
            bnb_4bit_use_double_quant=True,
        )

    print(f"📦 Loading model (transformers): {model_id} (4-bit={load_4bit})")
    model = AutoModelForCausalLM.from_pretrained(model_id, **load_kwargs)

    served_name = model_id
    if adapter_id:
        from peft import PeftModel
        print(f"📦 Loading LoRA adapter (safetensors): {adapter_id}")
        model = PeftModel.from_pretrained(model, adapter_id)
        served_name = adapter_id
        print(f"  ✅ Adapter loaded: {adapter_id}")

    model.eval()
    print(f"  ✅ Model ready! Serving as '{served_name}'")
    return model, tokenizer, served_name

def setup_ngrok(port: int, token: str = None):
    """Tạo ngrok tunnel, trả về public URL."""
    from pyngrok import ngrok

    auth_token = token or os.environ.get("NGROK_AUTH_TOKEN")
    if not auth_token:
        print("❌ Cần ngrok token! Đăng ký miễn phí: https://ngrok.com")
        print("   Dùng --ngrok-token <token> hoặc set env NGROK_AUTH_TOKEN")
        return None

    ngrok.set_auth_token(auth_token)
    tunnel = ngrok.connect(port)
    public_url = tunnel.public_url
    return public_url

def create_app(model, tokenizer, served_name: str):
    """Tạo FastAPI app với OpenAI-compatible endpoints."""
    import torch
    from fastapi import FastAPI
    from pydantic import BaseModel as PydanticModel

    app = FastAPI(title="EduBoost Cloud Eval Server")

    class Message(PydanticModel):
        role: str
        content: str

    class ChatRequest(PydanticModel):
        model: str = served_name
        messages: List[Message]
        max_tokens: Optional[int] = 512
        temperature: Optional[float] = 0.0

    class ChatChoice(PydanticModel):
        index: int = 0
        message: Message
        finish_reason: str = "stop"

    class ChatResponse(PydanticModel):
        id: str
        object: str = "chat.completion"
        created: int
        model: str
        choices: List[ChatChoice]

    @app.get("/v1/models")
    def list_models():
        return {
            "object": "list",
            "data": [{"id": served_name, "object": "model"}],
        }

    @app.post("/v1/chat/completions")
    def chat_completions(request: ChatRequest):
        messages = [{"role": m.role, "content": m.content} for m in request.messages]

        if hasattr(tokenizer, "apply_chat_template"):
            input_text = tokenizer.apply_chat_template(
                messages, tokenize=False, add_generation_prompt=True,
            )
        else:
            input_text = "\n".join(f"{m['role']}: {m['content']}" for m in messages)
            input_text += "\nassistant: "

        inputs = tokenizer(input_text, return_tensors="pt").to(model.device)

        with torch.no_grad():
            outputs = model.generate(
                **inputs,
                max_new_tokens=request.max_tokens,
                temperature=max(request.temperature, 0.01),
                do_sample=request.temperature > 0,
                pad_token_id=tokenizer.pad_token_id,
            )

        new_tokens = outputs[0][inputs["input_ids"].shape[1]:]
        response_text = tokenizer.decode(new_tokens, skip_special_tokens=True).strip()

        return ChatResponse(
            id=f"chatcmpl-{uuid.uuid4().hex[:8]}",
            created=int(time.time()),
            model=served_name,
            choices=[ChatChoice(message=Message(role="assistant", content=response_text))],
        )

    return app
def main():
    
    print(f"{'━'*60}")
    print(f"  ⚙️  CONFIG")
    print(f"     MODEL_ID:    {MODEL_ID}")
    print(f"     ADAPTER_ID:  {ADAPTER_ID or '(none — base model only)'}")
    print(f"     LOAD_4BIT:   {LOAD_4BIT}")

    # B1: Load model
    model, tokenizer, served_name = load_model(
        model_id=MODEL_ID,
        adapter_id=ADAPTER_ID,
        load_4bit=LOAD_4BIT,
    )

    app = create_app(model, tokenizer, served_name)

    public_url = setup_ngrok(PORT, NGROK_TOKEN)

    print(f"\n{'━'*60}")
    print(f"  🌐 Local:  http://0.0.0.0:{PORT}/v1")
    if public_url:
        print(f"  🔗 Public: {public_url}/v1")
    print(f"  🤖 Model:  {served_name}")
    print(f"{'━'*60}")

    if public_url:
        print(f"\n  Dùng với run_eval:")
        print(f"  python -m training.evaluation.run_eval generate --eval 0 --model 0 \\")
        print(f"      --url {public_url}/v1\n")

    # B4: Chạy server
    import threading
    import uvicorn
    import nest_asyncio
    import time

    # Đảm bảo Notebook đã được áp dụng bản vá vòng lặp sự kiện
    nest_asyncio.apply()

    def start_uvicorn():
        # Buộc sử dụng loop="asyncio" để đồng bộ với nest_asyncio trên Python 3.12+
        uvicorn.run(app, host="0.0.0.0", port=PORT, loop="asyncio")

    # Khởi chạy server trên một luồng phụ chạy ngầm
    server_thread = threading.Thread(target=start_uvicorn, daemon=True)
    server_thread.start()
    
    print(f"🚀 Server đã khởi động thành công trong Luồng phụ ngầm (Thread ID: {server_thread.ident})")
    time.sleep(2)
